cmake_minimum_required(VERSION 3.24)
project(NZRdspDexed VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This renderer currently targets Windows x64")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(DEXED_SOURCE_ROOT "" CACHE PATH "Pinned external Dexed source checkout")
if(NOT EXISTS "${DEXED_SOURCE_ROOT}/source/PluginProcessor.cpp")
  message(FATAL_ERROR "Pass DEXED_SOURCE_ROOT")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${DEXED_SOURCE_ROOT}" rev-parse HEAD OUTPUT_VARIABLE dexed_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT dexed_revision STREQUAL "fea2f0f0f2c397799f59b27eb11c038cd34b22c5")
  message(FATAL_ERROR "Dexed source revision differs from the pinned revision")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${DEXED_SOURCE_ROOT}" status --porcelain --untracked-files=no OUTPUT_VARIABLE dexed_dirty OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT dexed_dirty STREQUAL "")
  message(FATAL_ERROR "Preserve tracked Dexed/submodule changes before building")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${DEXED_SOURCE_ROOT}" submodule status --recursive OUTPUT_VARIABLE dexed_submodules COMMAND_ERROR_IS_FATAL ANY)
if(dexed_submodules MATCHES "(^|\\n)[+U-]")
  message(FATAL_ERROR "Initialize the exact pinned Dexed submodules")
endif()
set(DEXED_JUCE_PATH "${DEXED_SOURCE_ROOT}/libs/JUCE" CACHE PATH "" FORCE)
set(DEXED_SKIP_VST3 ON CACHE BOOL "" FORCE)
add_subdirectory("${DEXED_SOURCE_ROOT}" "${CMAKE_BINARY_DIR}/dexed" EXCLUDE_FROM_ALL)

# Disable desktop initialization in a build copy; keep the DSP implementation intact.
file(READ "${DEXED_SOURCE_ROOT}/source/PluginProcessor.cpp" processor)
foreach(call IN ITEMS "resolvAppDir();" "loadPreference();" "setupStartupCart();" "mtsClient = MTS_RegisterClient();")
  string(FIND "${processor}" "${call}" position)
  if(position EQUAL -1)
    message(FATAL_ERROR "Missing expected constructor call: ${call}")
  endif()
  string(REPLACE "${call}" "/* NZRdsp offline: ${call} disabled */" processor "${processor}")
endforeach()
string(REPLACE "setCurrentProgram(0);" "applyProgram(Program::initialProgram());" processor "${processor}")
file(WRITE "${CMAKE_BINARY_DIR}/PluginProcessor-offline.cpp" "${processor}")
get_target_property(dexed_sources Dexed SOURCES)
list(FILTER dexed_sources EXCLUDE REGEX "^PluginProcessor\\.cpp$")
set_property(TARGET Dexed PROPERTY SOURCES "${dexed_sources}")
target_sources(Dexed PRIVATE "${CMAKE_BINARY_DIR}/PluginProcessor-offline.cpp")
target_include_directories(Dexed PRIVATE "${DEXED_SOURCE_ROOT}/source/parameter")

juce_add_console_app(nzrdsp-dexed-native PRODUCT_NAME "NZRdsp Dexed renderer")
target_sources(nzrdsp-dexed-native PRIVATE main.cpp)
target_compile_features(nzrdsp-dexed-native PRIVATE cxx_std_17)
target_include_directories(nzrdsp-dexed-native PRIVATE
  "${DEXED_SOURCE_ROOT}/source"
  "${DEXED_SOURCE_ROOT}/libs"
  "${DEXED_SOURCE_ROOT}/libs/MTS-ESP/Client"
  "${CMAKE_BINARY_DIR}/dexed/source/Dexed_artefacts/JuceLibraryCode")
target_compile_definitions(nzrdsp-dexed-native PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=0)
target_link_libraries(nzrdsp-dexed-native PRIVATE Dexed DexedResources clap_juce_extensions tuning-library juce::juce_audio_utils)
